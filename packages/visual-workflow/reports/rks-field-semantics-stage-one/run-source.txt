import fs from 'node:fs';
import path from 'node:path';
import {spawn} from 'node:child_process';
import {pathToFileURL} from 'node:url';
import crypto from 'node:crypto';
const dir='packages/visual-workflow/reports/rks-field-semantics-stage-one';
const history=JSON.parse(fs.readFileSync('packages/visual-workflow/reports/rks-semantic-scope-and-operations/test-results.json'));
const [label,suite]=process.argv.slice(2);
const prior=history.results.find(x=>x.name===suite);
let args=prior?.args.slice();
if(suite==='paired')args=['--import','tsx','--test','--test-reporter=tap','--test-concurrency=1','packages/visual-workflow/tests/restaurant-knowledge/spatial-semantics/field-contract.test.ts'];
if(!args)throw Error('UNKNOWN_SUITE');
if(suite==='e1e-probes')args[args.length-1]=path.resolve(dir,label+'-probes.json');
if(suite==='workflow'&&label.startsWith('after'))args.push('packages/visual-workflow/tests/restaurant-knowledge/spatial-semantics/field-contract.test.ts');
const env=Object.fromEntries(Object.entries(process.env).filter(([k])=>!/(?:API_?KEY|SECRET|TOKEN|CREDENTIAL|PASSWORD|MASTERPIECE_|OPENAI|ANTHROPIC|DASHSCOPE|VOLCENGINE|QWEN)/iu.test(k)));
env.NODE_OPTIONS='--import='+pathToFileURL(path.resolve('.codex-smoke/rks-field-semantics-stage-one/network-off.mjs')).href;
env.npm_config_offline='true';env.npm_config_audit='false';env.npm_config_fund='false';
const started=Date.now(),file=dir+'/'+label+'.txt';const log=fs.createWriteStream(file,{flags:'wx'});
const child=spawn(process.execPath,args,{env,stdio:['ignore','pipe','pipe']});
child.stdout.pipe(log,{end:false});child.stderr.pipe(log,{end:false});
child.on('error',e=>{console.error(e.message);process.exitCode=1;});
child.on('close',code=>log.end(()=>{
 const body=fs.readFileSync(file,'utf8'),matches=[...body.matchAll(/^(ok|not ok) \d+ - (.+)$/gm)];
 const cases=matches.map((m,i)=>({title:m[2],status:m[1]==='ok'?'pass':'fail',...(m[1]==='not ok'?{diagnostic:body.slice(m.index,matches[i+1]?.index??body.length)}:{})}));
 const counts=Object.fromEntries([...body.matchAll(/^# (tests|pass|fail|cancelled|skipped) (\d+)/gm)].map(m=>[m[1],Number(m[2])]));
 const record={label,suite,args,exitCode:code,durationMs:Date.now()-started,counts,cases,log:file,sha256:crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')};
 fs.writeFileSync(dir+'/'+label+'.json',JSON.stringify(record,null,2),{flag:'wx'});
 console.log(JSON.stringify({label,exitCode:code,counts,failed:cases.filter(x=>x.status==='fail').map(x=>x.title)}));
}));
