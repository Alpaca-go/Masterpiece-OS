import http from 'node:http';
import https from 'node:https';
import net from 'node:net';
import tls from 'node:tls';
import dns from 'node:dns';
import dgram from 'node:dgram';
import {syncBuiltinESMExports} from 'node:module';
const deny=()=>{throw Error('FIELD_SEMANTICS_OFFLINE_NETWORK_FORBIDDEN');};
const original=net.Socket.prototype.connect;
net.Socket.prototype.connect=function(...args){
 const options=Array.isArray(args[0])?args[0]:args;
 const name=typeof options[0]==='string'?options[0]:options[0]?.path;
 return typeof name==='string'&&/^\\\\[.?]\\pipe\\/u.test(name)?original.apply(this,args):deny();
};
globalThis.fetch=deny;globalThis.WebSocket=class{constructor(){deny();}};
http.get=http.request=https.get=https.request=tls.connect=deny;
dns.lookup=dns.resolve=dgram.createSocket=deny;
syncBuiltinESMExports();
